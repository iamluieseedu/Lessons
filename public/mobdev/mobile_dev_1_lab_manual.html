Step 1: Create a New Project in Android Studio

Open Android Studio Arctic Fox (2020.3.1) and follow these 4 simple clicks.

1
Open Android Studio and click "New Project"

Look at the window that opens. Under the Phone and Tablet tab, click on "Empty Activity". Then click the blue Next button.

2
Type these exact settings into the boxes:

• Name: CampusConnect
• Package name: com.example.campusconnect
• Language: Kotlin (make sure it says Kotlin, not Java!)
• Minimum SDK: API 21: Android 5.0 (Lollipop)

3
Click the blue "Finish" button!

Now take a sip of water and wait 1 to 2 minutes. Watch the bottom right bar finish loading. When it stops loading, your project is ready!

Check your screen!

On the left side of your screen, you should now see a folder tree with app. You are ready for Step 2!

MainActivity.kt 
package com.example.campusconnect

import android.content.Intent
import androidx.appcompat.app.AppCompatActivity
import android.os.Bundle
import android.widget.Button

class MainActivity : AppCompatActivity() {

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)

        // 1. Find the 3 buttons
        val btnEnterDashboard = findViewById<Button>(R.id.btnEnterDashboard)
        val btnLogin = findViewById<Button>(R.id.btnLogin)
        val btnSignUp = findViewById<Button>(R.id.btnSignUp)

        // 2. Click button ➔ Open Dashboard!
        btnEnterDashboard.setOnClickListener {
            val intent = Intent(this, DashboardActivity::class.java)
            startActivity(intent)
        }

        // 3. Click button ➔ Open Login Screen!
        btnLogin.setOnClickListener {
            val intent = Intent(this, LoginActivity::class.java)
            startActivity(intent)
        }

        // 4. Click button ➔ Open Sign Up Screen!
        btnSignUp.setOnClickListener {
            val intent = Intent(this, SignUpActivity::class.java)
            startActivity(intent)
        }
    }
}

DashboardActivity.kt 
package com.example.campusconnect

import android.content.Intent
import androidx.appcompat.app.AppCompatActivity
import android.os.Bundle
import android.widget.Button
import android.widget.LinearLayout
import android.widget.Toast

class DashboardActivity : AppCompatActivity() {

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_dashboard)

        // 1. Find cards and buttons
        val cardProfile = findViewById<LinearLayout>(R.id.cardProfile)
        val cardCourses = findViewById<LinearLayout>(R.id.cardCourses)
        val cardSettings = findViewById<LinearLayout>(R.id.cardSettings)
        val btnReturnHome = findViewById<Button>(R.id.btnReturnHome)
        val btnSwitchToLogin = findViewById<Button>(R.id.btnSwitchToLogin)

        // 2. Card click toasts (little popup messages!)
        cardProfile.setOnClickListener {
            Toast.makeText(this, "Profile: IT Student", Toast.LENGTH_SHORT).show()
        }

        cardCourses.setOnClickListener {
            Toast.makeText(this, "Enrolled: Mob Dev 1 Lab", Toast.LENGTH_SHORT).show()
        }

        cardSettings.setOnClickListener {
            Toast.makeText(this, "Settings: Arctic Fox 2020.3.1", Toast.LENGTH_SHORT).show()
        }

        // 3. Return to Welcome Screen
        btnReturnHome.setOnClickListener {
            val intent = Intent(this, MainActivity::class.java)
            intent.flags = Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TASK
            startActivity(intent)
        }

        // 4. Switch to Login Screen
        btnSwitchToLogin.setOnClickListener {
            startActivity(Intent(this, LoginActivity::class.java))
        }
    }
}

package com.example.campusconnect

import android.content.Intent
import androidx.appcompat.app.AppCompatActivity
import android.os.Bundle
import android.widget.Button
import android.widget.Toast

class LoginActivity : AppCompatActivity() {

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_login)

        val btnLogin = findViewById<Button>(R.id.btnPerformLogin)
        val btnGoToSignUp = findViewById<Button>(R.id.btnGoToSignUp)
        val btnBackToWelcome = findViewById<Button>(R.id.btnBackToWelcome)

        // 1. Tapping Log In opens Dashboard immediately!
        btnLogin.setOnClickListener {
            Toast.makeText(this, "Navigating to Dashboard...", Toast.LENGTH_SHORT).show()
            val intent = Intent(this, DashboardActivity::class.java)
            startActivity(intent)
        }

        // 2. Tapping Sign Up opens Sign Up screen
        btnGoToSignUp.setOnClickListener {
            val intent = Intent(this, SignUpActivity::class.java)
            startActivity(intent)
        }

        // 3. Tapping Back returns to Welcome
        btnBackToWelcome.setOnClickListener {
            finish()
        }
    }
}

Where to go 📁 app > java > com.example.campusconnect > SignUpActivity.kt

SignUpActivity.kt 
package com.example.campusconnect

import android.content.Intent
import androidx.appcompat.app.AppCompatActivity
import android.os.Bundle
import android.widget.Button
import android.widget.Toast

class SignUpActivity : AppCompatActivity() {

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_sign_up)

        val btnSignUp = findViewById<Button>(R.id.btnPerformSignUp)
        val btnGoToLogin = findViewById<Button>(R.id.btnGoToLogin)
        val btnBack = findViewById<Button>(R.id.btnBackFromSignUp)

        // 1. Tapping Register opens Dashboard immediately!
        btnSignUp.setOnClickListener {
            Toast.makeText(this, "Registration Complete! Welcome!", Toast.LENGTH_SHORT).show()
            val intent = Intent(this, DashboardActivity::class.java)
            startActivity(intent)
        }

        // 2. Tapping Log In opens Login screen
        btnGoToLogin.setOnClickListener {
            val intent = Intent(this, LoginActivity::class.java)
            startActivity(intent)
        }

        // 3. Tapping Back returns to Welcome
        btnBack.setOnClickListener {
            finish()
        }
    }
}

Oops! Got a Red Line? (5-Second Fixes!)

Here are the easiest fixes for the most common little accidents.

Red line on the letter "R"?

Happens when Android Studio gets confused.

Easy Fix: Look at the very top menu of Android Studio. Click Build ➔ click Clean Project. Wait 10 seconds. Done!
App crashes when clicking a button?

Usually means an Activity was not registered.

Easy Fix: Go to Step 7. Copy the AndroidManifest.xml code again and paste it. That fixes 100% of crashes!
Red line on "logo"?

Means the logo file is missing in drawable.

Easy Fix: Go to Step 2.4. Right-click drawable ➔ New ➔ Drawable Resource File ➔ name it logo ➔ paste our code!
I see code instead of design preview?

You are in "Code only" view.

Easy Fix: Look at the top right of the code window. Click the icon that says Split or Design to see the phone screen!
